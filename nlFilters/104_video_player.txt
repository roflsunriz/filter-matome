# nlフィルタ定義(文字コード判定用なのでこの行は削除しないこと)

#キャッシュ済みの有料動画(視聴期限切れ)などをコメント付きで視聴するための機能。
#正しいIDを動画ファイルに指定し、cacheフォルダに置くことで任意の動画に任意のコメントを被せて
#視聴することも可能。
#mlink-video-controller内のモジュールのdeleted-video-detectorで削除済み動画を検知し、
#その動画をこのプレイヤーで再生する機能も有する。
#Requires common.es.js
[Replace]
Name = 有料動画でのキャッシュ利用（公式動画ページ用）
FullURL = https?://www\.nicovideo\.jp\/watch\/.*?
Match<
(\r\n)?(</head>)
>
Replace<
$1
<!--- 有料動画でのキャッシュ利用（公式動画ページ用） --->
<script type="module" src="/local/features/dist/video-player.es.js"></script>
<!--- End of 有料動画でのキャッシュ利用（公式動画ページ用） --->
$2
>

#これは必須 無いと再生できない
[Replace]
Name = isNeedPaymentステート書き換え（公式動画ページ用）
FullURL = https?:\/\/www\.nicovideo\.jp\/watch\/(?:so)?[0-9]{1,12}
EachLine = TRUE
Multi = TRUE
ContentType = text/html
Match<
((&quot;)okReason\2:\2).*?(\2)
>
Replace<
$1PURELY$3
>
